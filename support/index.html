const form = document.getElementById('support-form'); form.onsubmit = async (e) => { e.preventDefault(); const btn = form.querySelector('button'); const originalText = btn.innerText; btn.innerText = 'Отправка...'; btn.disabled = true; const formData = new FormData(form); try { const response = await fetch(form.action, { method: 'POST', body: formData, headers: { 'Accept': 'application/json' } }); if (response.ok) { alert('Спасибо! Ваше обращение успешно отправлено на почту поддержки Решётки.'); form.reset(); } else { alert('Произошла ошибка при отправке. Проверьте правильность введенных данных.'); } } catch (error) { alert('Не удалось связаться с сервером. Проверьте интернет-соединение.'); } finally { btn.innerText = originalText; btn.disabled = false; } };
